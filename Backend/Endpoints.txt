# User

Create User > POST "/user"
Req: { email: string, name: string, password: string }
Res: { id: string, email: string, name: string, createdAt: Date }

# Auth

Login > POST "/auth"
Req: { email: string, password: string }
Res: { success: true }

# Folder

Create Folder > POST "/folder"
Req: { userId: string, name: string }
Res: { id: string, userId: string, name: string }
